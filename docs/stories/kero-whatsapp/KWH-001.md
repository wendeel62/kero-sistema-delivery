# KWH-001: Modelo de Dados PostgreSQL — WhatsApp

## Definição
Criar o esquema completo de banco de dados para o sistema WhatsApp multi-tenant do Kero, com tabelas para dispositivos, conversas, mensagens e contatos.

## Modelo de Dados
```sql
-- Extensão UUID
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- devices: instâncias WhatsApp por usuário
CREATE TABLE devices (
  id            UUID        PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id        UUID       NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  device_name   VARCHAR(100) NOT NULL,
  instance_name VARCHAR(100) NOT NULL,  -- nome no Evolution API
  phone_number  VARCHAR(20)  DEFAULT '',
  avatar_url    TEXT         DEFAULT '',
  state         VARCHAR(20)  DEFAULT 'pending',  -- pending | connecting | open | close | paired
  evo_instance_id VARCHAR(100) DEFAULT '',
  qr_code       TEXT         DEFAULT '',
  qr_updated_at TIMESTAMPTZ  DEFAULT NULL,
  created_at    TIMESTAMPTZ  DEFAULT NOW(),
  updated_at    TIMESTAMPTZ  DEFAULT NOW(),
  UNIQUE(user_id, device_name)
);

-- conversations: conversas do WhatsApp
CREATE TABLE conversations (
  id              UUID        PRIMARY KEY DEFAULT uuid_generate_v4(),
  device_id       UUID        NOT NULL REFERENCES devices(id) ON DELETE CASCADE,
  jid            VARCHAR(255) NOT NULL,  -- WhatsApp JID (ex: 5511999999999@s.whatsapp.net)
  name           VARCHAR(255) DEFAULT '',
  pushname       VARCHAR(255) DEFAULT '',
  avatar_url     TEXT         DEFAULT '',
  is_group      BOOLEAN       DEFAULT FALSE,
  is_archived   BOOLEAN      DEFAULT FALSE,
  is_muted      BOOLEAN      DEFAULT FALSE,
  is_pinned     BOOLEAN      DEFAULT FALSE,
  unread_count  INTEGER      DEFAULT 0,
  last_message_id UUID       DEFAULT NULL,
  last_message_at TIMESTAMPTZ  DEFAULT NULL,
  created_at     TIMESTAMPTZ  DEFAULT NOW(),
  updated_at     TIMESTAMPTZ  DEFAULT NOW(),
  UNIQUE(device_id, jid)
);

-- messages: mensagens do WhatsApp
CREATE TABLE messages (
  id              UUID        PRIMARY KEY DEFAULT uuid_generate_v4(),
  device_id       UUID        NOT NULL REFERENCES devices(id) ON DELETE CASCADE,
  conversation_id UUID      NOT NULL REFERENCES conversations(id) ON DELETE CASCADE,
  message_id      VARCHAR(255) NOT NULL,  -- ID do Evolution API
  evo_id          VARCHAR(255) DEFAULT '',
  direction       VARCHAR(4)   NOT NULL,  -- in | out
  content         TEXT         DEFAULT '',
  content_type   VARCHAR(50)  DEFAULT 'text',  -- text | image | video | audio | document | sticker | location | contact | reaction
  media_url      TEXT         DEFAULT '',
  media_mime     VARCHAR(100) DEFAULT '',
  media_size    BIGINT      DEFAULT 0,
  from_jid      VARCHAR(255) DEFAULT '',
  to_jid        VARCHAR(255) DEFAULT '',
  from_me      BOOLEAN      DEFAULT FALSE,
  from_ BOT     BOOLEAN      DEFAULT FALSE,
  timestamp     TIMESTAMPTZ  NOT NULL,
  expires_at    TIMESTAMPTZ  DEFAULT NULL,
  is_deleted   BOOLEAN      DEFAULT FALSE,
  is_forwarded BOOLEAN      DEFAULT FALSE,
  reply_to_id   UUID        DEFAULT NULL REFERENCES messages(id) ON DELETE SET NULL,
  reaction     VARCHAR(100) DEFAULT '',
  reactions    JSONB       DEFAULT '[]',
  created_at   TIMESTAMPTZ  DEFAULT NOW()
);

-- contacts: contatos do WhatsApp
CREATE TABLE contacts (
  id         UUID       PRIMARY KEY DEFAULT uuid_generate_v4(),
  device_id  UUID      NOT NULL REFERENCES devices(id) ON DELETE CASCADE,
  jid       VARCHAR(255) NOT NULL,
  name      VARCHAR(255) DEFAULT '',
  pushname  VARCHAR(255) DEFAULT '',
  avatar_url TEXT      DEFAULT '',
  about     TEXT      DEFAULT '',
  isblocked BOOLEAN   DEFAULT FALSE,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(device_id, jid)
);

-- indexes para performance
CREATE INDEX idx_devices_user_id ON devices(user_id);
CREATE INDEX idx_conversations_device_id ON conversations(device_id);
CREATE INDEX idx_conversations_last_message_at ON conversations(last_message_at DESC);
CREATE INDEX idx_messages_device_id ON messages(device_id);
CREATE INDEX idx_messages_conversation_id ON messages(conversation_id);
CREATE INDEX idx_messages_timestamp ON messages(timestamp DESC);
CREATE INDEX idx_messages_conversation_timestamp ON messages(conversation_id, timestamp DESC);
CREATE INDEX idx_messages_from_jid ON messages(from_jid);
CREATE INDEX idx_messages_to_jid ON messages(to_jid);
CREATE INDEX idx_contacts_device_id ON contacts(device_id);

-- triggers para updated_at automático
CREATE OR REPLACE FUNCTION update_updated_at()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN NEW.updated_at = NOW(); RETURN NEW; END; $$;

CREATE TRIGGER devices_updated_at BEFORE UPDATE ON devices FOR EACH ROW EXECUTE PROCEDURE update_updated_at();
CREATE TRIGGER conversations_updated_at BEFORE UPDATE ON conversations FOR EACH ROW EXECUTE PROCEDURE update_updated_at();
CREATE TRIGGER contacts_updated_at BEFORE UPDATE ON contacts FOR EACH ROW EXECUTE PROCEDURE update_updated_at();
```
